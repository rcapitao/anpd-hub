<script>
(function () {
  var list = document.getElementById("noticias-lista");
  var tagsList = document.getElementById("noticias-filtro-tags");
  var countEl = document.getElementById("noticias-contagem");
  var clearBtn = document.getElementById("noticias-limpar");
  if (!list || !tagsList) return;

  var cards = list.querySelectorAll(".anpd-card");
  var chips = tagsList.querySelectorAll("[data-tag]");
  var active = new Set();

  function render() {
    var visible = 0;
    cards.forEach(function (card) {
      var tags = (card.getAttribute("data-tags") || "").split(",");
      var show = active.size === 0 || tags.some(function (t) { return active.has(t); });
      card.hidden = !show;
      if (show) visible += 1;
    });
    chips.forEach(function (chip) {
      var tag = chip.getAttribute("data-tag");
      var isActive = active.has(tag);
      chip.classList.toggle("is-active", isActive);
      chip.setAttribute("aria-pressed", isActive ? "true" : "false");
    });
    if (countEl) {
      countEl.textContent = visible + " de " + cards.length + " notícias";
    }
    if (clearBtn) {
      clearBtn.hidden = active.size === 0;
    }
  }

  chips.forEach(function (chip) {
    chip.addEventListener("click", function () {
      var tag = chip.getAttribute("data-tag");
      if (active.has(tag)) {
        active.delete(tag);
      } else {
        active.add(tag);
      }
      render();
    });
  });

  if (clearBtn) {
    clearBtn.addEventListener("click", function () {
      active.clear();
      render();
    });
  }

  render();
})();
</script>
