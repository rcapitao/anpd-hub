<script>
(function () {
  var container = document.getElementById("noticias-filtros");
  var table = document.getElementById("noticias-tabela");
  var countEl = document.getElementById("noticias-contagem");
  if (!container || !table) return;

  var chips = container.querySelectorAll("[data-tag]");
  var rows = table.querySelectorAll("tbody tr");
  var active = new Set();

  function render() {
    var visible = 0;
    rows.forEach(function (row) {
      var tags = (row.getAttribute("data-tags") || "").split(",");
      var show = active.size === 0 || tags.some(function (t) { return active.has(t); });
      row.hidden = !show;
      if (show) visible += 1;
    });
    chips.forEach(function (chip) {
      var tag = chip.getAttribute("data-tag");
      var isActive = tag === "all" ? active.size === 0 : active.has(tag);
      chip.classList.toggle("is-active", isActive);
      chip.setAttribute("aria-pressed", isActive ? "true" : "false");
    });
    if (countEl) {
      countEl.textContent = visible + " de " + rows.length + " notícias";
    }
  }

  chips.forEach(function (chip) {
    chip.addEventListener("click", function () {
      var tag = chip.getAttribute("data-tag");
      if (tag === "all") {
        active.clear();
      } else if (active.has(tag)) {
        active.delete(tag);
      } else {
        active.add(tag);
      }
      render();
    });
  });

  render();
})();
</script>
